(async function downloadEachPageIndividually(){
  const start = 1;
  const end = 13;
  const base = 'https://ruouchuan.vn/collections/whiskey?page=';
  const delayMs = 800; // delay giữa các request

  for(let p = start; p <= end; p++){
    const url = base + p + '/';
    console.log(`Fetching page ${p}:`, url);
    try{
      const resp = await fetch(url, { credentials: 'same-origin' });
      if(!resp.ok) throw new Error(`HTTP ${resp.status}`);
      const text = await resp.text();

      // Tạo file HTML và tải ngay
      const blob = new Blob([text], { type: 'text/html;charset=utf-8' });
      const filename = `hedon-page-${p}.html`;
      const a = document.createElement('a');
      a.href = URL.createObjectURL(blob);
      a.download = filename;
      document.body.appendChild(a);
      a.click();
      a.remove();
      URL.revokeObjectURL(a.href);

      console.log(`✅ Downloaded page ${p} as ${filename}`);
    } catch(err) {
      console.error(`❌ Failed to fetch page ${p}:`, err);
    }
    await new Promise(r => setTimeout(r, delayMs));
  }
  console.log('Hoàn tất tải từ page', start, 'đến', end);
})();
